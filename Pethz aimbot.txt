local function getHum(p)
    local c = p and p.Character
    return c and c:FindFirstChildOfClass("Humanoid")
end

local function isAlive(p)
    local h = getHum(p)
    return h and h.Health > 0
end

local function getPart(p, n)
    local c = p and p.Character
    return c and (c:FindFirstChild(n) or c:FindFirstChild("HumanoidRootPart"))
end

local function sameTeam(p)
    if not Aim.TeamCheck then return false end
    return p.Team ~= nil and LP.Team ~= nil and p.Team == LP.Team
end

local rayParams = RaycastParams.new()
rayParams.FilterType = Enum.RaycastFilterType.Exclude
rayParams.IgnoreWater = true

local function isVisible(part)
    if not Aim.WallCheck then return true end
    if not part then return false end
    rayParams.FilterDescendantsInstances = { LP.Character, Cam }
    local origin = Cam.CFrame.Position
    local dir = part.Position - origin
    local r = workspace:Raycast(origin, dir, rayParams)
    if r then
        return r.Instance:IsDescendantOf(part.Parent)
    end
    return true
end

local function applyConfigData(data)
    if type(data) ~= "table" then return end
    if type(data.Aim) == "table" then
        for k, v in pairs(data.Aim) do Aim[k] = v end
    end
    if type(data.ESP) == "table" then
        for k, v in pairs(data.ESP) do
            if k ~= "ColorName" then ESP[k] = v end
        end
        if data.ESP.ColorName and Themes[data.ESP.ColorName] then
            CurrentESPColorName = data.ESP.ColorName
            ESPColor = Themes[data.ESP.ColorName]
        end
    end
    if type(data.Mov) == "table" then
        for k, v in pairs(data.Mov) do Mov[k] = v end
    end
    if type(data.NoRecoil) == "boolean" then NoRecoil = data.NoRecoil end
    if type(data.NoSpread) == "boolean" then NoSpread = data.NoSpread end
    if type(data.AutoReload) == "boolean" then AutoReload = data.AutoReload end
    if type(data.NoTexture) == "boolean" and NoTexture ~= data.NoTexture then
        NoTexture = data.NoTexture
        toggleTextures(NoTexture)
    end
    if type(data.ShowFPS) == "boolean" then ShowFPS = data.ShowFPS end
    if type(data.CarFly) == "boolean" then CarFly = data.CarFly end
    if type(data.RespawnEnabled) == "boolean" then Respawn.Enabled = data.RespawnEnabled end
    if type(data.CameraFOV) == "number" then CameraFOV = data.CameraFOV end
end

local function saveConfig()
    local snap = serializeConfig()
    SavedConfigSnapshot = snap
    if writefile then
        pcall(function()
            writefile(CONFIG_FILE, HttpService:JSONEncode(snap))
        end)
    end
end

local function loadConfig()
    if readfile and isfile and pcall(function() return isfile(CONFIG_FILE) end) then
        local ok, raw = pcall(function() return readfile(CONFIG_FILE) end)
        if ok and raw and raw ~= "" then
            local decOk, decoded = pcall(function() return HttpService:JSONDecode(raw) end)
            if decOk and type(decoded) == "table" then
                SavedConfigSnapshot = decoded
                applyConfigData(decoded)
                return true
            end
        end
    end
    if SavedConfigSnapshot then
        applyConfigData(SavedConfigSnapshot)
        return true
    end
    return false
end

local function isPlayerOwned(inst)
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Character and inst:IsDescendantOf(p.Character) then
            return true
        end
        local bp = p:FindFirstChildOfClass("Backpack")
        if bp and inst:IsDescendantOf(bp) then
            return true
        end
    end
    return false
end

local function getGrabbableItems()
    local items = {}
    local seen = {}

    for _, obj in ipairs(workspace:GetDescendants()) do
        if not isPlayerOwned(obj) then
            if obj:IsA("Tool") and not seen[obj] then
                seen[obj] = true
                local handle = obj:FindFirstChild("Handle") or obj:FindFirstChildWhichIsA("BasePart", true)
                table.insert(items, {
                    name = obj.Name,
                    kind = "Tool",
                    instance = obj,
                    part = handle
                })
            elseif obj:IsA("ProximityPrompt") and obj.Parent and obj.Parent:IsA("BasePart") and not seen[obj.Parent] then
                seen[obj.Parent] = true
                local label = (obj.ActionText ~= "" and obj.ActionText) or obj.Parent.Name
                table.insert(items, {
                    name = label,
                    kind = "Prompt",
                    instance = obj,
                    part = obj.Parent
                })
            elseif obj:IsA("ClickDetector") and obj.Parent and obj.Parent:IsA("BasePart") and not seen[obj.Parent] then
                seen[obj.Parent] = true
                table.insert(items, {
                    name = obj.Parent.Name,
                    kind = "Click",
                    instance = obj,
                    part = obj.Parent
                })
            end
        end
    end

    return items
end

local function grabItem(entry)
    local char = LP.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local backpack = LP:FindFirstChildOfClass("Backpack")
    if not entry or not entry.instance or not entry.instance.Parent then return end

    if entry.kind == "Tool" then
        local tool = entry.instance
        if tool:IsA("Tool") then
            local handle = tool:FindFirstChild("Handle") or tool:FindFirstChildWhichIsA("BasePart", true)
            if handle and root then
                local origCF = handle.CFrame
                local origAnchored = handle.Anchored
                pcall(function()
                    handle.Anchored = false
                    handle.CanCollide = false
                    handle.CFrame = root.CFrame * CFrame.new(0, 0, -2)
                end)
                if firetouchinterest then
                    pcall(function()
                        firetouchinterest(root, handle, 0)
                        task.wait(0.05)
                        firetouchinterest(root, handle, 1)
                    end)
                end
                task.delay(0.15, function()
                    if tool.Parent ~= char and tool.Parent ~= backpack and backpack then
                        pcall(function() tool.Parent = backpack end)
                    end
                    if handle and handle.Parent and tool.Parent ~= char and tool.Parent ~= backpack then
                        pcall(function()
                            handle.CFrame = origCF
                            handle.Anchored = origAnchored
                        end)
                    end
                end)
            elseif backpack then
                pcall(function() tool.Parent = backpack end)
            end
        end
    elseif entry.kind == "Prompt" then
        local prompt = entry.instance
        local part = entry.part
        if root and part then
            local origCF = root.CFrame
            pcall(function()
                root.CFrame = part.CFrame * CFrame.new(0, 0, 2)
            end)
            task.wait(0.06)
            if fireproximityprompt then
                pcall(function() fireproximityprompt(prompt) end)
            else
                pcall(function() prompt:InputHoldBegin() end)
                task.wait(math.clamp(prompt.HoldDuration or 0.1, 0.05, 0.6))
                pcall(function() prompt:InputHoldEnd() end)
            end
            task.delay(0.12, function()
                if root and root.Parent then
                    pcall(function() root.CFrame = origCF end)
                end
            end)
        end
    elseif entry.kind == "Click" then
        local cd = entry.instance
        if fireclickdetector then
            pcall(function() fireclickdetector(cd) end)
        end
    end
end

function toggleTextures(remove)
    textureGen += 1
    local gen = textureGen

    if remove then
        task.spawn(function()
            for _, obj in ipairs(workspace:GetDescendants()) do
                if gen ~= textureGen then return end
                local isPlayerPart = false
                local parent = obj.Parent
                while parent do
                    if parent:IsA("Model") and parent:FindFirstChildOfClass("Humanoid") then
                        isPlayerPart = true
                        break
                    end
                    parent = parent.Parent
                end
                if not isPlayerPart then
                    if obj:IsA("BasePart") then
                        if not originalMaterials[obj] then
                            originalMaterials[obj] = { Material = obj.Material, Color = obj.Color }
                        end
                        pcall(function()
                            obj.Material = Enum.Material.SmoothPlastic
                            obj.Color = Color3.fromRGB(185, 190, 200)
                        end)
                    elseif obj:IsA("Decal") or obj:IsA("Texture") then
                        if not hiddenTextures[obj] then
                            hiddenTextures[obj] = obj.Transparency
                        end
                        pcall(function() obj.Transparency = 1 end)
                    end
                end
            end
        end)
    else
        task.spawn(function()
            for obj, data in pairs(originalMaterials) do
                if gen ~= textureGen then return end
                if obj and obj.Parent then
                    pcall(function()
                        obj.Material = data.Material
                        obj.Color = data.Color
                    end)
                end
            end
            table.clear(originalMaterials)

            for obj, trans in pairs(hiddenTextures) do
                if gen ~= textureGen then return end
                if obj and obj.Parent then
                    pcall(function() obj.Transparency = trans end)
                end
            end
            table.clear(hiddenTextures)
        end)
    end
end

local function setupRespawn()

        if Respawn.Enabled then
            local root = char:FindFirstChild("HumanoidRootPart")
            if root then Respawn.LastCFrame = root.CFrame end
        end
    end)
end

        if root then

local function getTarget()
    local best, bestD = nil, math.huge
    local center = Vector2.new(Cam.ViewportSize.X / 2, Cam.ViewportSize.Y / 2)
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and isAlive(p) and not sameTeam(p) then
            local part = getPart(p, Aim.Part)
            if part and isVisible(part) then
                local sp, on = Cam:WorldToViewportPoint(part.Position)
                if on then
                    local d = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                    if d <= Aim.FOV / 2 and d < bestD then
                        bestD = d
                        best = p
                    end
                end
            end
        end
    end
    return best
end

local function attachChams(p, char)
    if not char then return end
    local old = ESP.ChamsObjs[p]
    if old then pcall(function() old:Destroy() end) end
    local hl = Instance.new("Highlight")
    hl.Name = "PethzChams"
    hl.FillTransparency = 0.4
    hl.OutlineTransparency = 0
    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    hl.FillColor = ESPColor
    hl.OutlineColor = Color3.fromRGB(255, 255, 255)
    hl.Enabled = ESP.Enabled and ESP.Chams
    hl.Parent = char
    ESP.ChamsObjs[p] = hl
end

local function createESP(p)
    if p == LP then return end
    if not ESP.Objs[p] then
        local o = {}
        o.BT, o.BB, o.BL, o.BR = mk("Line"), mk("Line"), mk("Line"), mk("Line")
        o.Name = mk("Text")
        if o.Name then o.Name.Size = 13; o.Name.Center = true; o.Name.Outline = true end
        o.HBg = mk("Line")
        if o.HBg then o.HBg.Color = Color3.fromRGB(30, 30, 30); o.HBg.Thickness = 4 end
        o.HBar = mk("Line")
        if o.HBar then o.HBar.Thickness = 4 end
        o.Snap = mk("Line")
        if o.Snap then o.Snap.Thickness = 1.5 end
        o.HeadCircle = mk("Circle")
        if o.HeadCircle then
            o.HeadCircle.Thickness = 1.5
            o.HeadCircle.NumSides = 24
            o.HeadCircle.Filled = false
            o.HeadCircle.Visible = false
        end
        o.Skel = {}
        for i = 1, #BONES_R15 do
            local l = mk("Line")
            if l then l.Thickness = 1 end
            o.Skel[i] = l
        end
        ESP.Objs[p] = o
    end
    if p.Character then
        attachChams(p, p.Character)
    end
end

local function hideAll(o)
    for _, v in pairs(o) do
        if type(v) == "table" then
            for _, s in pairs(v) do if s then pcall(function() s.Visible = false end) end end
        elseif v then
            pcall(function() v.Visible = false end)
        end
    end
end

local function renderESP()
    if not ESP.Enabled then
        for _, o in pairs(ESP.Objs) do hideAll(o) end
        for _, hl in pairs(ESP.ChamsObjs) do if hl and hl.Parent then hl.Enabled = false end end
        return
    end

    for p, o in pairs(ESP.Objs) do
        local isTeam = ESP.TeamCheck and p.Team ~= nil and LP.Team ~= nil and p.Team == LP.Team
        local hl = ESP.ChamsObjs[p]
        if p == LP or not isAlive(p) or isTeam then
            hideAll(o)
            if hl and hl.Parent then hl.Enabled = false end
        else
            if hl and hl.Parent then
                hl.Enabled = ESP.Chams
                hl.FillColor = ESPColor
            end
            local c = p.Character
            local head = c and c:FindFirstChild("Head")
            local root = c and c:FindFirstChild("HumanoidRootPart")
            local hum = getHum(p)
            if not (head and root and hum) then
                hideAll(o)
            else
                local hs, hon = Cam:WorldToViewportPoint(head.Position)
                local rs, ron = Cam:WorldToViewportPoint(root.Position)
                if not (hon and ron) then
                    hideAll(o)
                else
                    local hp, rp = Vector2.new(hs.X, hs.Y), Vector2.new(rs.X, rs.Y)
                    local height = math.max((hp - rp).Magnitude * 1.6, 10)
                    local width = height * 0.45
                    local top, bot = hp.Y - height * 0.3, rp.Y + height * 0.3
                    local lft, rgt = rp.X - width / 2, rp.X + width / 2

                    if ESP.Box and o.BT then
                        o.BT.Color, o.BB.Color, o.BL.Color, o.BR.Color = ESPColor, ESPColor, ESPColor, ESPColor
                        o.BT.From, o.BT.To = Vector2.new(lft, top), Vector2.new(rgt, top)
                        o.BB.From, o.BB.To = Vector2.new(lft, bot), Vector2.new(rgt, bot)
                        o.BL.From, o.BL.To = Vector2.new(lft, top), Vector2.new(lft, bot)
                        o.BR.From, o.BR.To = Vector2.new(rgt, top), Vector2.new(rgt, bot)
                        for _, l in ipairs({ o.BT, o.BB, o.BL, o.BR }) do l.Visible = true end
                    else
                        for _, l in ipairs({ o.BT, o.BB, o.BL, o.BR }) do if l then l.Visible = false end end
                    end

                    if ESP.Name and o.Name then
                        o.Name.Color = ESPColor
                        local dist = math.floor((root.Position - Cam.CFrame.Position).Magnitude)
                        o.Name.Text = p.Name .. " [" .. dist .. "m]"
                        o.Name.Position = Vector2.new(
                            rp.X,
                            ESP.Pos == "Top" and (hp.Y - height * 0.5 - 22)
                                or ESP.Pos == "Bottom" and (rp.Y + height * 0.5 + 8)
                                or hp.Y
                        )
                        o.Name.Visible = true
                    elseif o.Name then
                        o.Name.Visible = false
                    end

                    if ESP.Health and o.HBar then
                        local pct = math.clamp(hum.Health / math.max(hum.MaxHealth, 1), 0, 1)
                        local bx = rgt + 8
                        o.HBg.From, o.HBg.To = Vector2.new(bx, top), Vector2.new(bx, bot)
                        o.HBg.Visible = true
                        o.HBar.From, o.HBar.To = Vector2.new(bx, bot - (bot - top) * pct), Vector2.new(bx, bot)
                        o.HBar.Color = ESPColor
                        o.HBar.Visible = true
                    else
                        if o.HBar then o.HBar.Visible = false end
                        if o.HBg then o.HBg.Visible = false end
                    end

                    if ESP.Line and o.Snap then
                        o.Snap.Color = ESPColor
                        o.Snap.From = Vector2.new(
                            Cam.ViewportSize.X / 2,
                            ESP.Pos == "Top" and 0 or ESP.Pos == "Center" and Cam.ViewportSize.Y / 2 or Cam.ViewportSize.Y
                        )
                        o.Snap.To = Vector2.new(rp.X, rp.Y)
                        o.Snap.Visible = true
                    elseif o.Snap then
                        o.Snap.Visible = false
                    end

                    if ESP.Skeleton then
                        local isR15 = c:FindFirstChild("UpperTorso") ~= nil
                        local boneList = isR15 and BONES_R15 or BONES_R6
                        for i, pair in ipairs(boneList) do
                            local a, b, line = c:FindFirstChild(pair[1]), c:FindFirstChild(pair[2]), o.Skel[i]
                            if line and a and b then
                                line.Color = ESPColor
                                local as, aon = Cam:WorldToViewportPoint(a.Position)
                                local bs, bon = Cam:WorldToViewportPoint(b.Position)
                                if aon and bon then
                                    line.From, line.To = Vector2.new(as.X, as.Y), Vector2.new(bs.X, bs.Y)
                                    line.Visible = true
                                else
                                    line.Visible = false
                                end
                            elseif line then
                                line.Visible = false
                            end
                        end
                        for i = #boneList + 1, #o.Skel do
                            if o.Skel[i] then o.Skel[i].Visible = false end
                        end

                        if o.HeadCircle then
                            local topHeadScreen = Cam:WorldToViewportPoint(head.Position + Vector3.new(0, head.Size.Y * 0.5, 0))
                            local botHeadScreen = Cam:WorldToViewportPoint(head.Position - Vector3.new(0, head.Size.Y * 0.5, 0))
                            local rad = math.clamp(math.abs(topHeadScreen.Y - botHeadScreen.Y) * 0.65, 3, 28)
                            o.HeadCircle.Position = hp
                            o.HeadCircle.Radius = rad
                            o.HeadCircle.Color = ESPColor
                            o.HeadCircle.Visible = true
                        end
                    else
                        for _, l in ipairs(o.Skel) do if l then l.Visible = false end end
                        if o.HeadCircle then o.HeadCircle.Visible = false end
                    end
                end
            end
        end
    end
end

local bhopCooldown = false
local fpsAccum = 0
local fpsFrames = 0

trackConn(RunService.RenderStepped:Connect(function(dt)
    Cam = workspace.CurrentCamera or Cam

    if ShowFPS then
        fpsAccum += dt
        fpsFrames += 1
        if fpsAccum >= 0.35 then
            local curFps = math.floor((fpsFrames / fpsAccum) + 0.5)
            fpsLabel.Text = "⚡ FPS: " .. tostring(curFps)
            fpsAccum = 0
            fpsFrames = 0
        end
    end

    if Cam and Cam.FieldOfView ~= CameraFOV then
        Cam.FieldOfView = CameraFOV
    end

    if fovCircle then
        fovCircle.Position = Vector2.new(Cam.ViewportSize.X / 2, Cam.ViewportSize.Y / 2)
        fovCircle.Radius = Aim.FOV / 2
        fovCircle.Color = ESPColor
        fovCircle.Visible = Aim.Enabled and Aim.ShowFOV
    end

    if Aim.Enabled then
        local target = getTarget()
        if target then
            local part = getPart(target, Aim.Part)
            if part then
                local predictedPos = part.Position + (part.AssemblyLinearVelocity * Aim.Prediction)
                local targetCF = CFrame.new(Cam.CFrame.Position, predictedPos)
                if Aim.Smooth > 0 then
                    Cam.CFrame = Cam.CFrame:Lerp(targetCF, math.clamp(1 - Aim.Smooth, 0.02, 1))
                else
                    Cam.CFrame = targetCF
                end
            end
        end
    elseif NoRecoil then
        local cf = Cam.CFrame
        local look = cf.LookVector
        if look.Y > 0.01 then
            local adjustedLook = Vector3.new(look.X, math.max(look.Y - 0.012, -0.99), look.Z).Unit
            Cam.CFrame = CFrame.lookAt(cf.Position, cf.Position + adjustedLook)
        end
    end

    renderESP()
end))

trackConn(RunService.Heartbeat:Connect(function(dt)
    local char = LP.Character
    local hum = getHum(LP)
    local root = char and char:FindFirstChild("HumanoidRootPart")

    if Mov.Spin and root then
        root.CFrame = root.CFrame * CFrame.Angles(0, math.rad(Mov.SpinSpeed) * (dt * 60), 0)
    end

    if hum then
        if Mov.Speed then
            hum.WalkSpeed = Mov.SpeedVal
        elseif not Mov.Bhop and hum.WalkSpeed ~= 16 then
            hum.WalkSpeed = 16
        end

        if Mov.Jump then
            hum.UseJumpPower = true
            hum.JumpPower = Mov.JumpVal
        elseif hum.JumpPower ~= 50 then
            hum.JumpPower = 50
        end

        if Mov.Bhop and root and hum.FloorMaterial ~= Enum.Material.Air and hum.MoveDirection.Magnitude > 0.1 and not bhopCooldown then
            bhopCooldown = true
            root.AssemblyLinearVelocity = Vector3.new(root.AssemblyLinearVelocity.X, 50, root.AssemblyLinearVelocity.Z)
            if hum.WalkSpeed < 35 then hum.WalkSpeed = 35 end
            task.delay(0.2, function() bhopCooldown = false end)
        end

        if CarFly and hum.SeatPart then
            local seat = hum.SeatPart
            local vehicle = seat.Parent
            local vRoot = vehicle and (vehicle:FindFirstChild("Chassis") or vehicle:FindFirstChild("Base") or vehicle:FindFirstChild("Body")) or seat
            if vRoot and vRoot:IsA("BasePart") then
                local moveDir = hum.MoveDirection
                local up = 0
                if UIS:IsKeyDown(Enum.KeyCode.Space) or hum.Jump then
                    up = 1
                elseif UIS:IsKeyDown(Enum.KeyCode.LeftControl) then
                    up = -1
                end
                if moveDir.Magnitude > 0.1 or up ~= 0 then
                    local speed = 80
                    local targetPos = vRoot.Position + (moveDir * speed * dt) + Vector3.new(0, up * speed * dt, 0)
                    local targetCF = CFrame.new(targetPos, targetPos + Cam.CFrame.LookVector)
                    vRoot.CFrame = vRoot.CFrame:Lerp(targetCF, 0.3)
                    vRoot.AssemblyLinearVelocity = Vector3.zero
                end
            end
        end
    end
end))
